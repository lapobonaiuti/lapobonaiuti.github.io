<script>
    const container = document.getElementById('graph-container');
    const Graph = ForceGraph3D()(container)
        .backgroundColor('#1a1a24') 
        .nodeRelSize(5)
        .nodeColor(n => n.inGiant ? '#c95a49' : '#888888') // Modificato per usare il rosso-terracotta del tuo tema
        .linkColor(l => l.inGiant ? '#c95a49' : '#555555')
        .linkWidth(l => l.inGiant ? 1.5 : 0.5)
        .linkOpacity(0.7);

    function updateGraph() {
        const n = parseInt(document.getElementById('n-slider').value);
        const p = parseFloat(document.getElementById('p-slider').value);
        
        document.getElementById('n-val').innerText = n;
        document.getElementById('p-val').innerText = p.toFixed(3);
        
        const threshold = 1 / (n - 1);
        document.getElementById('stat-p').innerText = p.toFixed(3);
        document.getElementById('stat-threshold').innerText = threshold.toFixed(3);

        const nodes = Array.from({length: n}, (_, i) => ({id: i}));
        const links = [];
        for (let i = 0; i < n; i++) {
            for (let j = i + 1; j < n; j++) {
                if (Math.random() < p) {
                    links.push({source: i, target: j});
                }
            }
        }

        const adj = Array.from({length: n}, () => []);
        links.forEach(l => {
            adj[l.source].push(l.target);
            adj[l.target].push(l.source);
        });

        const visited = new Array(n).fill(false);
        const components = [];

        for (let i = 0; i < n; i++) {
            if (!visited[i]) {
                const comp = [];
                const queue = [i];
                visited[i] = true;
                while(queue.length > 0) {
                    const curr = queue.shift();
                    comp.push(curr);
                    adj[curr].forEach(neighbor => {
                        if(!visited[neighbor]) {
                            visited[neighbor] = true;
                            queue.push(neighbor);
                        }
                    });
                }
                components.push(comp);
            }
        }

        const giant = components.reduce((max, c) => c.length > max.length ? c : max, []);
        const giantSet = new Set(giant);
        document.getElementById('stat-giant').innerText = giant.length + " nodi";

        nodes.forEach(n => n.inGiant = giantSet.has(n.id));
        links.forEach(l => {
            l.inGiant = giantSet.has(l.source) && giantSet.has(l.target);
        });

        Graph.graphData({ nodes: nodes, links: links });
    }

    document.getElementById('n-slider').addEventListener('input', updateGraph);
    document.getElementById('p-slider').addEventListener('input', updateGraph);

    const wrapper = document.getElementById('graph-wrapper');
    const btn = document.getElementById('fullscreen-btn');

    function toggleFullScreen() {
        if (!document.fullscreenElement) {
            wrapper.requestFullscreen().catch(err => {
                alert(`Errore: Impossibile attivare lo schermo intero (${err.message})`);
            });
        } else {
            document.exitFullscreen();
        }
    }

    document.addEventListener('fullscreenchange', () => {
        if (document.fullscreenElement) {
            btn.innerText = "⛶ Esci";
            Graph.width(window.innerWidth).height(window.innerHeight);
        } else {
            btn.innerText = "⛶ Schermo Intero";
            Graph.width(wrapper.clientWidth).height(500);
        }
    });

    updateGraph();
    
    setTimeout(() => {
        Graph.width(wrapper.clientWidth).height(500);
    }, 100);
</script>
